# DWG is exchanged with the external LibreDWG programs, never by linking libredwg.
# The default-on bridge searches tools/libredwg/ beside the executable, then PATH.
set(PD_DWG_READER "")
set(PD_DWG_WRITER "")
if(NOT DEFINED PD_LIBREDWG_VERSION)
    set(PD_LIBREDWG_VERSION "unknown")
endif()
if(PD_ENABLE_LIBREDWG)
    if(TARGET dwg2dxf)
        find_program(PD_DWG2DXF_EXECUTABLE NAMES dwg2dxf HINTS "${libredwg_BINARY_DIR}" NO_DEFAULT_PATH)
        find_program(PD_DXF2DWG_EXECUTABLE NAMES dxf2dwg HINTS "${libredwg_BINARY_DIR}" NO_DEFAULT_PATH)
    endif()
    if(NOT PD_DWG2DXF_EXECUTABLE)
        find_program(PD_DWG2DXF_EXECUTABLE NAMES dwg2dxf DOC "Absolute path to trusted external reader")
    endif()
    if(NOT PD_DXF2DWG_EXECUTABLE)
        find_program(PD_DXF2DWG_EXECUTABLE NAMES dxf2dwg DOC "Absolute path to trusted external writer")
    endif()
    if(PD_DWG2DXF_EXECUTABLE)
        set(PD_DWG_READER "${PD_DWG2DXF_EXECUTABLE}")
    endif()
    if(PD_DXF2DWG_EXECUTABLE)
        set(PD_DWG_WRITER "${PD_DXF2DWG_EXECUTABLE}")
    endif()
endif()
configure_file(src/dwg_paths.hpp.in dwg_paths.hpp @ONLY)
add_library(pd_io STATIC src/registry.cpp src/unicode.cpp src/dxf_common.cpp src/dxf_reader.cpp
    src/dxf_writer.cpp src/dwg.cpp src/svg.cpp src/pdf.cpp src/pdf_font.cpp)
target_link_libraries(pd_io PUBLIC pd_core)
target_include_directories(pd_io PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(PD_ENABLE_LIBREDWG)
    target_sources(pd_io PRIVATE src/worker.cpp)
    target_compile_definitions(pd_io PRIVATE PD_ENABLE_LIBREDWG PD_LIBREDWG_VERSION="${PD_LIBREDWG_VERSION}")
    if(TARGET pd_dwg_tools)
        add_dependencies(pd_io pd_dwg_tools)
    endif()
endif()
add_executable(pdconvert convert.cpp)
target_link_libraries(pdconvert PRIVATE pd_io)
if(MINGW)
    target_link_options(pdconvert PRIVATE -municode)
endif()
